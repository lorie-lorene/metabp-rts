import requests, subprocess, re

# Login
s = requests.Session()
s.get("http://localhost:8080/api/v1/verifycode/generate")

    ["sudo", "docker", "logs", "--tail", "5",

    stderr=subprocess.STDOUT
).decode()

    headers={"Content-Type": "application/json",

print(f"Token OK: {token[:30]}...")

headers = {"Authorization": f"Bearer {token}", "Content-Type": "application/json"}

# Tester les endpoints
endpoints = [
    ("GET", "http://localhost:12345/api/v1/stationservice/stations"),
    ("GET", "http://localhost:14567/api/v1/trainservice/trains"),
    ("GET", "http://localhost:11178/api/v1/routeservice/routes"),
